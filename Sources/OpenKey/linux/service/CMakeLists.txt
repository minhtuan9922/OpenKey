if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The session service currently supports Linux only")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENKEY_SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
find_package(Threads REQUIRED)
add_library(openkey_fcitx_status STATIC FcitxStatus.cpp InputMonitor.cpp TrayItem.cpp X11Hotkey.cpp SettingsLaunch.cpp)
if(OPENKEY_BUILD_X11_HOTKEY)
    pkg_check_modules(OPENKEY_XCB REQUIRED IMPORTED_TARGET xcb)
    target_compile_definitions(openkey_fcitx_status PRIVATE OPENKEY_HAVE_X11_HOTKEY=1)
    target_link_libraries(openkey_fcitx_status PRIVATE PkgConfig::OPENKEY_XCB)
endif()
target_link_libraries(openkey_fcitx_status PUBLIC PkgConfig::OPENKEY_SYSTEMD Threads::Threads)
target_include_directories(openkey_fcitx_status PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(openkey_fcitx_status PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(openkey-service main.cpp)
target_link_libraries(openkey-service PRIVATE openkey_fcitx_status)
target_compile_options(openkey-service PRIVATE -Wall -Wextra -Wpedantic)
if(OPENKEY_WARNINGS_AS_ERRORS)
    target_compile_options(openkey-service PRIVATE -Werror)
endif()
install(TARGETS openkey-service RUNTIME DESTINATION bin COMPONENT openkey_service)
install(PROGRAMS "${PROJECT_SOURCE_DIR}/scripts/openkey-tray-autostart.sh"
    DESTINATION bin RENAME openkey-tray-autostart COMPONENT openkey_service)
install(PROGRAMS "${PROJECT_SOURCE_DIR}/scripts/openkey-tray-systemd.sh"
    DESTINATION bin RENAME openkey-tray-systemd COMPONENT openkey_service)
if(OPENKEY_BUILD_SMOKE_TESTS)
    find_program(OPENKEY_BASH bash REQUIRED)
    add_executable(openkey_hotkey_policy_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxHotkeyPolicyTests.cpp)
    target_link_libraries(openkey_hotkey_policy_tests PRIVATE openkey_fcitx_status)
    target_compile_options(openkey_hotkey_policy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(NOT OPENKEY_BUILD_X11_HOTKEY)
        target_compile_definitions(openkey_hotkey_policy_tests PRIVATE OPENKEY_TEST_NO_X11_HOTKEY=1)
    endif()
    add_test(NAME openkey_hotkey_policy_tests COMMAND openkey_hotkey_policy_tests)
    if(OPENKEY_BUILD_X11_HOTKEY)
        find_program(OPENKEY_XVFB Xvfb)
        if(OPENKEY_XVFB)
            pkg_check_modules(OPENKEY_X11_TEST REQUIRED IMPORTED_TARGET x11 xtst)
            add_executable(openkey_x11_hotkey_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxX11HotkeyTests.cpp)
            target_link_libraries(openkey_x11_hotkey_tests PRIVATE openkey_fcitx_status PkgConfig::OPENKEY_X11_TEST)
            target_compile_options(openkey_x11_hotkey_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
            add_test(NAME openkey_x11_hotkey_tests COMMAND "${OPENKEY_BASH}"
                "${PROJECT_SOURCE_DIR}/Tests/RunX11HotkeyTests.sh" "${OPENKEY_XVFB}" "$<TARGET_FILE:openkey_x11_hotkey_tests>")
            set_tests_properties(openkey_x11_hotkey_tests PROPERTIES TIMEOUT 15)
        else()
            message(WARNING "Xvfb missing: X11 integration test unavailable; install xvfb to validate key grabs")
        endif()
    endif()
    add_test(NAME openkey_systemd_tests COMMAND "${OPENKEY_BASH}"
        "${PROJECT_SOURCE_DIR}/Tests/LinuxSystemdTests.sh"
        "${PROJECT_SOURCE_DIR}/scripts/openkey-tray-systemd.sh")
    set_tests_properties(openkey_systemd_tests PROPERTIES TIMEOUT 10)
    add_test(NAME openkey_autostart_tests COMMAND "${OPENKEY_BASH}"
        "${PROJECT_SOURCE_DIR}/Tests/LinuxAutostartTests.sh"
        "${PROJECT_SOURCE_DIR}/scripts/openkey-tray-autostart.sh")
    set_tests_properties(openkey_autostart_tests PROPERTIES TIMEOUT 10)
    add_test(NAME openkey_service_tests COMMAND "${OPENKEY_BASH}"
        "${PROJECT_SOURCE_DIR}/Tests/LinuxServiceTests.sh" "$<TARGET_FILE:openkey-service>")
    set_tests_properties(openkey_service_tests PROPERTIES TIMEOUT 20)
    find_program(OPENKEY_DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_executable(openkey_settings_launch_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxSettingsLaunchTests.cpp)
    target_link_libraries(openkey_settings_launch_tests PRIVATE openkey_fcitx_status)
    target_compile_options(openkey_settings_launch_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_settings_launch_tests COMMAND ${OPENKEY_DBUS_RUN_SESSION}
        --config-file=${PROJECT_SOURCE_DIR}/Tests/fixtures/dbus/no-activation.conf -- $<TARGET_FILE:openkey_settings_launch_tests>)
    set_tests_properties(openkey_settings_launch_tests PROPERTIES TIMEOUT 10)
    add_executable(openkey_fcitx_status_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxFcitxStatusTests.cpp)
    target_link_libraries(openkey_fcitx_status_tests PRIVATE openkey_fcitx_status Threads::Threads)
    target_compile_options(openkey_fcitx_status_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_fcitx_status_tests COMMAND ${OPENKEY_DBUS_RUN_SESSION}
        -- $<TARGET_FILE:openkey_fcitx_status_tests>)
    set_tests_properties(openkey_fcitx_status_tests PROPERTIES TIMEOUT 15)
    add_executable(openkey_input_monitor_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxInputMonitorTests.cpp)
    target_link_libraries(openkey_input_monitor_tests PRIVATE openkey_fcitx_status)
    target_compile_options(openkey_input_monitor_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_input_monitor_tests COMMAND openkey_input_monitor_tests)
    set_tests_properties(openkey_input_monitor_tests PROPERTIES TIMEOUT 15)
    add_executable(openkey_tray_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxTrayTests.cpp)
    target_link_libraries(openkey_tray_tests PRIVATE openkey_fcitx_status)
    target_compile_options(openkey_tray_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_tray_tests COMMAND ${OPENKEY_DBUS_RUN_SESSION}
        --config-file=${PROJECT_SOURCE_DIR}/Tests/fixtures/dbus/no-activation.conf -- $<TARGET_FILE:openkey_tray_tests>
        $<TARGET_FILE:openkey-service>)
    set_tests_properties(openkey_tray_tests PROPERTIES TIMEOUT 15)
endif()
