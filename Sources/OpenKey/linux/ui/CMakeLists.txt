if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The Qt settings preview currently targets Linux only")
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2 Network DBus)
if(Qt6_VERSION VERSION_GREATER_EQUAL "6.8")
    qt_policy(SET QTP0004 NEW)
endif()
set(CMAKE_AUTOMOC ON)
add_library(openkey_settings_model STATIC SettingsOverview.cpp SettingsOverview.h SettingsController.cpp SettingsController.h)
target_link_libraries(openkey_settings_model PUBLIC Qt6::Core Qt6::Network Qt6::DBus)
target_include_directories(openkey_settings_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(openkey_settings_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
qt_add_executable(openkey-settings main.cpp)
qt_add_qml_module(openkey-settings
    URI OpenKey.Settings VERSION 1.0 RESOURCE_PREFIX /qt/qml
    QML_FILES qml/Main.qml qml/UiPalette.qml qml/NavIcon.qml qml/NavButton.qml qml/InfoCard.qml qml/StyleSettings.qml qml/SettingsComboBox.qml)
target_link_libraries(openkey-settings PRIVATE Qt6::Quick Qt6::QuickControls2 openkey_settings_model)
# Shared Qt loads distro QML plugins at runtime; their static CMake targets
# are not necessarily shipped by the distro. Keep scanning for static Qt.
get_target_property(openkey_qt_type Qt6::Core TYPE)
if(openkey_qt_type STREQUAL "SHARED_LIBRARY")
    set_target_properties(openkey-settings PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
endif()
target_compile_options(openkey-settings PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS openkey-settings RUNTIME DESTINATION bin COMPONENT openkey_ui)
if(OPENKEY_BUILD_SMOKE_TESTS)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    find_program(OPENKEY_SETTINGS_DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_executable(openkey_settings_controller_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxSettingsControllerTests.cpp)
    target_link_libraries(openkey_settings_controller_tests PRIVATE openkey_settings_model Qt6::Test)
    target_compile_options(openkey_settings_controller_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_settings_controller_tests COMMAND openkey_settings_controller_tests)
    set_tests_properties(openkey_settings_controller_tests PROPERTIES TIMEOUT 30)
    add_executable(openkey_settings_model_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxSettingsModelTests.cpp)
    target_link_libraries(openkey_settings_model_tests PRIVATE openkey_settings_model Qt6::Test)
    target_compile_options(openkey_settings_model_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_settings_model_tests COMMAND openkey_settings_model_tests)
    set_tests_properties(openkey_settings_model_tests PROPERTIES TIMEOUT 15)
    qt_add_executable(openkey_settings_tests ${PROJECT_SOURCE_DIR}/Tests/LinuxSettingsTests.cpp)
    if(openkey_qt_type STREQUAL "SHARED_LIBRARY")
        set_target_properties(openkey_settings_tests PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
    endif()
    target_link_libraries(openkey_settings_tests PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Test openkey_settings_model)
    target_compile_definitions(openkey_settings_tests PRIVATE
        OPENKEY_SETTINGS_QML="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
    target_compile_options(openkey_settings_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME openkey_settings_tests COMMAND openkey_settings_tests)
    add_test(NAME openkey_settings_resource_smoke COMMAND openkey-settings --smoke-test)
    add_test(NAME openkey_settings_hidpi_tests COMMAND openkey_settings_tests)
    set_tests_properties(openkey_settings_tests openkey_settings_resource_smoke PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_IM_MODULE=compose"
        TIMEOUT 30)
    set_tests_properties(openkey_settings_hidpi_tests PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_IM_MODULE=compose;QT_SCALE_FACTOR=2"
        TIMEOUT 30)
endif()
